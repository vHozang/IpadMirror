function(padmirror_add_core_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE padmirror_core)
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

padmirror_add_core_test(packet_parser_test packet_parser/PacketParserTest.cpp)
padmirror_add_core_test(audio_ring_buffer_test ring_buffer/AudioRingBufferTest.cpp)
padmirror_add_core_test(quicktime_protocol_test protocol/QuickTimeProtocolTest.cpp)
padmirror_add_core_test(live_edge_test live_edge/LiveEdgeControllerTest.cpp)
