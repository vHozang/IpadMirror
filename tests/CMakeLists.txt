function(padmirror_add_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE padmirror_support)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /permissive- /EHsc /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

padmirror_add_test(receiver_config_test support/ReceiverConfigTest.cpp)
padmirror_add_test(receiver_identity_test support/ReceiverIdentityTest.cpp)
